// ============================================
// Schema: Gestión de Infraestructuras
// ============================================

// --- TABLAS DE REFERENCIA (ampliables sin migraciones) ---

Table roles {
  id          int       [pk, increment]
  name        varchar   [unique, not null, note: 'admin, editor, reader...']
  description text
  can_write   boolean   [not null, default: false]
  can_manage  boolean   [not null, default: false, note: 'Gestión de usuarios y configuración']
}

Table infra_statuses {
  id          int       [pk, increment]
  name        varchar   [unique, not null, note: 'active, inactive, maintenance...']
  description text
}

Table action_types {
  id              int       [pk, increment]
  name            varchar   [unique, not null, note: 'inspection, repair, installation...']
  description     text
  consumes_materials boolean [not null, default: false, note: 'Indica si este tipo de acción consume materiales']
}

// --- USUARIOS ---

Table users {
  id            int       [pk, increment]
  email         varchar   [unique, not null]
  password_hash varchar   [not null]
  full_name     varchar   [not null]
  role_id       int       [not null, ref: > roles.id]
  is_active     boolean   [not null, default: true]
  created_at    timestamp [not null, default: `now()`]
  updated_at    timestamp [not null, default: `now()`]
}

// --- INFRAESTRUCTURAS ---

Table infrastructures {
  id          int       [pk, increment]
  name        varchar   [not null]
  description text
  location    varchar
  status_id   int       [not null, ref: > infra_statuses.id]
  created_at  timestamp [not null, default: `now()`]
  updated_at  timestamp [not null, default: `now()`]
}

// --- ACCIONES ---

Table actions {
  id                int       [pk, increment]
  infrastructure_id int       [not null, ref: > infrastructures.id]
  performed_by      int       [not null, ref: > users.id]
  action_type_id    int       [not null, ref: > action_types.id]
  description       text
  performed_at      timestamp [not null, default: `now()`]
  created_at        timestamp [not null, default: `now()`]
}

// --- MATERIALES CONSUMIDOS ---
// No existe catálogo previo. Los materiales se registran
// en el momento en que se consumen en una acción.
// Cada fila es un material gastado en una acción concreta.

Table action_materials {
  id          int       [pk, increment]
  action_id   int       [not null, ref: > actions.id]
  name        varchar   [not null, note: 'Nombre del material consumido']
  description text
  unit        varchar   [not null, note: 'Unidad de medida: kg, m, unidades, litros...']
  quantity    decimal   [not null, note: 'Cantidad consumida']
  unit_cost   decimal   [note: 'Coste por unidad en el momento del consumo']
  total_cost  decimal   [note: 'quantity × unit_cost']
  supplier    varchar   [note: 'Proveedor del material']
  notes       text
}
