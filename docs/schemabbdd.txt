// ==================== AUTH / USERS ====================

Table roles {
  id          int         [pk, increment]
  name        varchar     [unique, not null]
  description varchar
  can_write   boolean     [not null, default: false]
  can_manage  boolean     [not null, default: false]
}

Table users {
  id                   int         [pk, increment]
  email                varchar     [unique, not null]
  password_hash        varchar     [not null]
  full_name            varchar     [not null]
  role_id              int         [not null, ref: > roles.id]
  is_active            boolean     [not null, default: true]
  must_change_password boolean     [not null, default: true]
  token_version        int         [not null, default: 0]
  created_at           timestamp   [not null, default: `now()`]
  updated_at           timestamp   [not null, default: `now()`]
  deleted_at           timestamp

  indexes {
    role_id
  }
}

// ==================== LOCATION HIERARCHY ====================

Table infrastructure_types {
  id          int       [pk, increment]
  name        varchar   [unique, not null]
  description varchar
  icon        varchar
  color       varchar
  deleted_at  timestamp
}

Table locations {
  id            int       [pk, increment]
  name          varchar   [not null]
  description   varchar
  path          varchar   [not null, default: '/', note: 'Materialized path: e.g., /1/4/']
  created_at    timestamp [not null, default: `now()`]
  updated_at    timestamp [not null, default: `now()`]
  deleted_at    timestamp
  latitude      float
  longitude     float
  image         varchar   [note: 'Path to location image']
  infra_type_id int       [not null, ref: > infrastructure_types.id]
  parent_id     int       [ref: > locations.id]

  indexes {
    parent_id
    path
    infra_type_id
  }
}

// ==================== MATERIAL CATALOG ====================

Table material_types {
  id                int       [pk, increment]
  code              varchar   [unique, not null]
  name              varchar   [not null]
  description       varchar
  icon              varchar
  created_at        timestamp [not null, default: `now()`]
  updated_at        timestamp [not null, default: `now()`]
  deleted_at        timestamp
  custom_attributes json      [not null, default: '[]', note: 'Technical attributes spec']
}

// Implicit many-to-many join table (Prisma-managed)
Table _infrastructure_type_to_material_type {
  a int [not null, ref: > infrastructure_types.id]
  b int [not null, ref: > material_types.id]

  indexes {
    (a, b) [pk]
  }
}

Table fixed_properties {
  id    int     [pk, increment]
  code  varchar [unique, not null]
  name  varchar [not null]
  type  varchar [not null, default: 'STRING']
}

// ==================== MATERIAL INSTANCES ====================

Table materials {
  id           int       [pk, increment]
  name         varchar   [not null]
  description  varchar
  installed_at timestamp [not null, default: `now()`]
  attributes   json      [not null, default: '{}']
  created_at   timestamp [not null, default: `now()`]
  updated_at   timestamp [not null, default: `now()`]
  deleted_at   timestamp
  type_id      int       [not null, ref: > material_types.id]
  location_id  int       [ref: > locations.id]

  indexes {
    type_id
    location_id
  }
}

// ==================== ACTIONS ====================

Table actions {
  id           int       [pk, increment]
  title        varchar   [not null]
  description  varchar
  performed_at timestamp [not null, default: `now()`]
  created_at   timestamp [not null, default: `now()`]
  updated_at   timestamp [not null, default: `now()`]
  location_id  int       [ref: > locations.id]
  latitude     float
  longitude    float
  performed_by int       [not null, ref: > users.id]

  indexes {
    location_id
    performed_by
    performed_at
  }
}

enum MaterialOperation {
  INSTALL
  UNINSTALL
  UPDATE
}

Table action_materials {
  action_id   int               [ref: > actions.id]
  material_id int               [ref: > materials.id]
  operation   MaterialOperation [not null]
  snapshot    json

  indexes {
    (action_id, material_id) [pk]
  }
}

Table system_settings {
  id    int     [pk, increment]
  key   varchar [unique, not null]
  value varchar [not null]
}

Table location_photos {
  id          int       [pk, increment]
  url         varchar   [not null]
  description varchar
  taken_at    timestamp [not null, default: `now()`]
  created_at  timestamp [not null, default: `now()`]
  location_id int       [not null, ref: > locations.id]
  action_id   int       [ref: > actions.id]

  indexes {
    location_id
    action_id
  }
}
