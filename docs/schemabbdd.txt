// control-actions — Database Schema (DBML)
// Fuente de verdad: backend/prisma/schema.prisma
// Actualizado: 2026-05-05

// ==================== AUTH / USERS ====================

Table roles {
  id          int         [pk, increment]
  name        varchar     [unique, not null]
  description varchar
  can_write   boolean     [not null, default: false]
  can_manage  boolean     [not null, default: false]
}

Table users {
  id                   int         [pk, increment]
  email                varchar     [not null]
  password_hash        varchar     [not null]
  full_name            varchar     [not null]
  role_id              int         [not null, ref: > roles.id]
  is_active            boolean     [not null, default: true]
  must_change_password boolean     [not null, default: true]
  created_at           timestamp   [not null, default: `now()`]
  updated_at           timestamp   [not null, default: `now()`]
  deleted_at           timestamp

  indexes {
    role_id
  }
}

// ==================== LOCATION HIERARCHY ====================
// Infrastructure → Dependency (árbol recursivo) → Structure → Material
// Un Material puede colgar directamente de Infrastructure, Dependency o Structure

// Nota: este DBML debe reflejar `backend/prisma/schema.prisma`.
// En Prisma actual, varios `code` son opcionales (nullable) aunque el backend puede exigirlos en el API.

Table infrastructures {
  id          int         [pk, increment]
  code        varchar
  name        varchar     [not null]
  description varchar
  created_at  timestamp   [not null, default: `now()`]
  updated_at  timestamp   [not null, default: `now()`]
  deleted_at  timestamp

  // Relación opcional con infrastructure_types (existe en Prisma)
  infra_type_id int       [ref: > infrastructure_types.id]
}

Table infrastructure_types {
  id          int       [pk, increment]
  name        varchar   [unique, not null]
  description varchar
  icon        varchar
  color       varchar
  deleted_at  timestamp
}

Table dependencies {
  id                 int       [pk, increment]
  code               varchar
  name               varchar   [not null]
  description        varchar
  infrastructure_id  int       [not null, ref: > infrastructures.id]
  parent_id          int       [ref: > dependencies.id]   // null = raíz
  created_at         timestamp [not null, default: `now()`]
  updated_at         timestamp [not null, default: `now()`]
  deleted_at         timestamp

  indexes {
    infrastructure_id
    parent_id
    (infrastructure_id, parent_id, code) [unique]
  }
}

Table structures {
  id                 int       [pk, increment]
  code               varchar
  name               varchar   [not null]
  description        varchar
  infrastructure_id  int       [ref: > infrastructures.id]   // nullable
  dependency_id      int       [ref: > dependencies.id]      // nullable
  created_at         timestamp [not null, default: `now()`]
  updated_at         timestamp [not null, default: `now()`]
  deleted_at         timestamp

  indexes {
    infrastructure_id
    dependency_id
  }
}

// ==================== MATERIAL CATALOG ====================

Table material_types {
  id          int       [pk, increment]
  code        varchar   [not null]
  name        varchar   [not null]
  description varchar
  icon        varchar
  created_at  timestamp [not null, default: `now()`]
  updated_at  timestamp [not null, default: `now()`]
  deleted_at  timestamp
}

// data_type: STRING | NUMBER | BOOLEAN | DATE | ENUM
Table material_categories {
  id               int       [pk, increment]
  code             varchar   [not null]
  name             varchar   [not null]
  description      varchar
  data_type        varchar   [not null, note: 'STRING | NUMBER | BOOLEAN | DATE | ENUM']
  unit             varchar
  required         boolean   [not null, default: false]
  sort_order       int       [not null, default: 0]
  enum_values      text[]
  validation       jsonb
  material_type_id int       [not null, ref: > material_types.id]

  indexes {
    material_type_id
    (material_type_id, code) [unique]
  }
}

// Material = instancia física de un MaterialType ubicada en la jerarquía.
// Exactamente uno de infrastructure_id / dependency_id / structure_id debe ser no nulo.
Table materials {
  id                 int       [pk, increment]
  code               varchar
  name               varchar   [not null]
  description        varchar
  serial_number      varchar
  installed_at       timestamp [not null, default: `now()`]
  attributes         jsonb     [not null, default: '{}', note: 'Validados contra material_categories del tipo']
  type_id            int       [not null, ref: > material_types.id]
  infrastructure_id  int       [ref: > infrastructures.id]
  dependency_id      int       [ref: > dependencies.id]
  structure_id       int       [ref: > structures.id]
  created_at         timestamp [not null, default: `now()`]
  updated_at         timestamp [not null, default: `now()`]
  deleted_at         timestamp

  indexes {
    type_id
    infrastructure_id
    dependency_id
    structure_id
  }
}

// ==================== ACTIONS ====================

Table action_types {
  id          int       [pk, increment]
  code        varchar   [not null]
  name        varchar   [not null]
  description varchar
  icon        varchar
  color       varchar
  deleted_at  timestamp
}

// Action = evento realizado sobre un Material concreto.
// No tiene soft-delete — es registro de auditoría inmutable.
Table actions {
  id           int       [pk, increment]
  title        varchar   [not null]
  description  varchar
  performed_at timestamp [not null, default: `now()`]
  created_at   timestamp [not null, default: `now()`]
  updated_at   timestamp [not null, default: `now()`]
  type_id      int       [not null, ref: > action_types.id]
  material_id  int       [not null, ref: > materials.id]
  performed_by int       [not null, ref: > users.id]

  indexes {
    material_id
    performed_by
    performed_at
  }
}
